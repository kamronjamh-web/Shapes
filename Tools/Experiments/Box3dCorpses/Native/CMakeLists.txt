cmake_minimum_required(VERSION 3.22)
project(box3d_corpse_bridge LANGUAGES C CXX)
# Clang's platform module rewrites archive rules during project(). Restore Zig's
# explicit subcommands here, before the upstream static target inherits them.
if(CMAKE_C_COMPILER_ARG1 MATCHES "^cc ")
    set(CMAKE_C_ARCHIVE_CREATE "<CMAKE_AR> ar qc <TARGET> <LINK_FLAGS> <OBJECTS>")
    set(CMAKE_C_ARCHIVE_FINISH "<CMAKE_AR> ranlib <TARGET>")
endif()
set(BOX3D_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../../.tmp/box3d-corpses/upstream" CACHE PATH "Pinned Box3D checkout")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BOX3D_PROFILE OFF CACHE BOOL "" FORCE)
set(BOX3D_VALIDATE OFF CACHE BOOL "" FORCE)
set(BOX3D_DOUBLE_PRECISION OFF CACHE BOOL "" FORCE)
add_subdirectory("${BOX3D_SOURCE_DIR}" box3d)
add_library(box3d_corpses SHARED corpse_bridge.c)
target_link_libraries(box3d_corpses PRIVATE box3d)
set_target_properties(box3d_corpses PROPERTIES C_STANDARD 17 PREFIX "")
add_executable(box3d_corpse_smoke smoke.c)
target_link_libraries(box3d_corpse_smoke PRIVATE box3d_corpses)
